const axios = require('axios')

async function createLicense(courseId, username, phone) {

    const response = await axios.post(
        'https://panel.spotplayer.ir/license/edit/',
        {
            course: [
                courseId
            ],
            name: username,
            watermark: {
                texts: [
                    {
                        text: phone
                    }
                ]
            }
        },
        {
            headers: {
                '$API': process.env.SPOT_API_KEY,
                '$LEVEL': '-1',
                'Content-Type': 'application/json'
            }
        }
    )

    return response.data
}
